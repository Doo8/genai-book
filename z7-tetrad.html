    from lfx.custom.custom_component.component import Component
from lfx.io import MessageTextInput, Output
from lfx.schema.message import Message


class CompensationCalc(Component):
    display_name = "Расчёт компенсации"
    description = "Считает компенсацию: 50 процентов от суммы, не больше 500 000 рублей."
    inputs = [MessageTextInput(name="amount", display_name="Сумма", tool_mode=True)]
    outputs = [Output(display_name="Результат", name="result", method="calculate")]

    def calculate(self) -> Message:
        raw = getattr(self, "amount", None)
        if raw is None or not str(raw).strip():
            return Message(text="Сумма не указана. Введите сумму в рублях, например: 966 687,60")
        cleaned = (str(raw).replace("\u00a0", "").replace(" ", "").replace(",", ".").strip())
        try:
            value = float(cleaned)
        except ValueError:
            return Message(text=f"Не удалось распознать сумму: «{raw}». Укажите число, например: 966 687,60")
        if value < 0:
            return Message(text="Сумма не может быть отрицательной.")
        compensation = round(value * 0.5, 2)
        if compensation > 500_000:
            compensation = 500_000.0
        return Message(text=f"{compensation:.2f}")

    нет суммыпонятное сообщение, а не ошибка
